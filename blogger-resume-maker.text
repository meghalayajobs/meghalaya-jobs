        match = trimmed.match(/^(\d{4})-(\d{2})-(\d{2})$/);
        if (match) {
          const year = Number(match[1]);
          const month = Number(match[2]) - 1;
          const day = Number(match[3]);
          const parsed = new Date(year, month, day);

          if (
            parsed.getFullYear() === year
            && parsed.getMonth() === month
            && parsed.getDate() === day
          ) {
            return parsed;
          }
        }

        return null;
      }

      function isSameDate(left, right) {
        return !!left
          && !!right
          && left.getFullYear() === right.getFullYear()
          && left.getMonth() === right.getMonth()
          && left.getDate() === right.getDate();
      }

      function renderCalendar() {
        if (!calendarState.input) return;

        const currentYear = new Date().getFullYear();
        const monthIndex = calendarState.visibleMonth.getMonth();
        const year = calendarState.visibleMonth.getFullYear();
        const firstDay = new Date(year, monthIndex, 1);
        const daysInMonth = new Date(year, monthIndex + 1, 0).getDate();
        const firstWeekday = firstDay.getDay();
        const startDate = new Date(year, monthIndex, 1 - firstWeekday);
        const selected = parseInputDate(calendarState.input.value);
        const today = new Date();
        const yearOptions = [];

        CALENDAR_MONTHS.forEach(function (monthName, index) {
          if (!calendarMonth.options[index]) {
            const option = document.createElement('option');
            option.value = String(index);
            option.textContent = monthName;
            calendarMonth.appendChild(option);
          }
        });

        for (let optionYear = currentYear + 20; optionYear >= currentYear - 100; optionYear -= 1) {
          yearOptions.push('<option value="' + optionYear + '"' + (optionYear === year ? ' selected' : '') + '>' + optionYear + '</option>');
        }

        calendarMonth.value = String(monthIndex);
        calendarYear.innerHTML = yearOptions.join('');
        calendarTitle.textContent = CALENDAR_MONTHS[monthIndex] + ' ' + year;

        const cells = [];
        for (let cellIndex = 0; cellIndex < 42; cellIndex += 1) {
          const cellDate = new Date(startDate.getFullYear(), startDate.getMonth(), startDate.getDate() + cellIndex);
          const isOutside = cellDate.getMonth() !== monthIndex;
          const classes = ['jmrm-calendar-day'];

          if (isOutside) classes.push('is-outside');
          if (isSameDate(cellDate, today)) classes.push('is-today');
          if (isSameDate(cellDate, selected)) classes.push('is-selected');

          cells.push(
            '<button type="button" class="' + classes.join(' ') + '"'
            + ' data-action="calendar-day"'
            + ' data-date="' + formatDate(cellDate) + '"'
            + ' aria-label="' + formatDate(cellDate) + '">'
            + cellDate.getDate()
            + '</button>'
          );
        }

        calendarGrid.innerHTML = cells.join('');
      }

      function normalizeWhitespace(value) {
        return String(value == null ? '' : value).replace(/\s+/g, ' ').trim();
      }

      function toPdfUpper(value) {
        return normalizeWhitespace(value).toUpperCase();
      }

      function toPdfTitleCase(value) {
        return normalizeWhitespace(value)
          .toLowerCase()
          .replace(/\b([a-z])/g, function (match) {
            return match.toUpperCase();
          });
      }

      function mapDivision(value) {
        const normalized = normalizeWhitespace(value).toLowerCase();
        if (normalized === '1st division') return 'First';
        if (normalized === '2nd division') return 'Second';
        if (normalized === '3rd division') return 'Third';
        return normalizeWhitespace(value);
      }

      function mapMaritalStatus(value) {
        const normalized = normalizeWhitespace(value).toLowerCase();
        if (normalized === 'unmarried') return 'SINGLE';
        if (normalized === 'married') return 'MARRIED';
        return normalized ? normalized.toUpperCase() : '';
      }

      function fitTextToWidth(pdf, text, maxWidth, fontStyle, fontSize, minSize) {
        const cleanText = normalizeWhitespace(text);
        let size = fontSize;

        pdf.setFont('times', fontStyle);
        pdf.setFontSize(size);

        while (cleanText && pdf.getTextWidth(cleanText) > maxWidth && size > minSize) {
          size = Math.max(minSize, Number((size - 0.2).toFixed(1)));
          pdf.setFontSize(size);
        }

        let fittedText = cleanText;
        if (fittedText && pdf.getTextWidth(fittedText) > maxWidth) {
          const ellipsis = '...';
          while (fittedText && pdf.getTextWidth(fittedText + ellipsis) > maxWidth) {
            fittedText = fittedText.slice(0, -1);
          }
          fittedText = fittedText ? fittedText + ellipsis : ellipsis;
        }

        return {
          text: fittedText,
          size: size
        };
      }

      function drawFittedText(pdf, text, x, y, maxWidth, fontStyle, fontSize, minSize, options) {
        const fitted = fitTextToWidth(pdf, text, maxWidth, fontStyle, fontSize, minSize);
        pdf.setFont('times', fontStyle);
        pdf.setFontSize(fitted.size);
        pdf.text(fitted.text || '', x, y, options || {});
      }

          { align: 'left' }
        );
        drawFittedText(
          pdf,
          'Email: ' + normalizeWhitespace(data.email),
          PDF_LAYOUT.contentLeft,
          PDF_LAYOUT.contact2Baseline,
          PDF_LAYOUT.photoX - PDF_LAYOUT.contentLeft - 12,
          'normal',
          11.2,
          9.6,
          { align: 'left' }
        );

        pdf.setFillColor(244, 241, 237);
        pdf.roundedRect(PDF_LAYOUT.photoX, PDF_LAYOUT.photoY, PDF_LAYOUT.photoWidth, PDF_LAYOUT.photoHeight, 1.5, 1.5, 'F');
        if (photoDataUrl) {
          pdf.addImage(photoDataUrl, 'JPEG', PDF_LAYOUT.photoX, PDF_LAYOUT.photoY, PDF_LAYOUT.photoWidth, PDF_LAYOUT.photoHeight, undefined, 'FAST');
        }
        pdf.setDrawColor(PDF_LAYOUT.accent[0], PDF_LAYOUT.accent[1], PDF_LAYOUT.accent[2]);
        pdf.setLineWidth(1.5);
        pdf.roundedRect(PDF_LAYOUT.photoX, PDF_LAYOUT.photoY, PDF_LAYOUT.photoWidth, PDF_LAYOUT.photoHeight, 1.5, 1.5, 'S');

        let nextTop = drawObjectiveSection(pdf, PDF_LAYOUT.firstSectionTop);
        const educationRows = buildEducationRowsForPdf(data.educationRows, notes);
        nextTop = drawEducationSection(pdf, nextTop, educationRows);
        const otherRows = buildOtherRowsForPdf(data.otherRows);
        if (otherRows.length) {
          nextTop = drawOtherQualificationsSection(pdf, nextTop, otherRows);
        }
        nextTop = drawPersonalInfoSection(pdf, nextTop, data);
        drawDeclarationSection(pdf, nextTop, data);

        if (normalizeWhitespace(data.experience)) {
          notes.push('Experience is not printed in this exact sample layout.');
        }

        return {
          blob: pdf.output('blob'),
          notes: notes
        };
      }

      function scheduleRender() {
        invalidateGeneratedPdf();
      }

      function clearErrors() {
        REQUIRED_FIELDS.forEach(function (id) {
          const field = document.getElementById(id);
          if (field) field.classList.remove('is-error');
        });
        photoInput.classList.remove('is-error');
        declarationCard.classList.remove('is-error');
      }

      function isEmailValid(value) {
        return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value);
      }

      function isMobileValid(value) {
        return /^\d{10}$/.test(value);
      }

      function validateForm() {
        clearErrors();
        clearStatus();

        let valid = true;

        REQUIRED_FIELDS.forEach(function (id) {
          const field = document.getElementById(id);
          if (!field || !field.value.trim()) {
            if (field) field.classList.add('is-error');
            valid = false;
          }
        });

        if (!photoDataUrl && !(photoInput.files && photoInput.files[0])) {
          photoInput.classList.add('is-error');
          valid = false;
        }

        if (!declarationCheckbox.checked) {
          declarationCard.classList.add('is-error');
          valid = false;
        }

        const email = getFieldValue('jmrm-email');
        const mobile = getFieldValue('jmrm-mobile');

        if (email && !isEmailValid(email)) {
          document.getElementById('jmrm-email').classList.add('is-error');
          valid = false;
        }

        if (mobile && !isMobileValid(mobile)) {
          document.getElementById('jmrm-mobile').classList.add('is-error');
          valid = false;
        }

        if (!valid) {
          setStatus('Please complete all required fields, upload a photo, tick the declaration checkbox, and check the mobile number and email format.', 'is-error');
        }

        return valid;
      }

      function waitForImageLoad(img) {
        if (img.decode) return img.decode().catch(function () { return null; });
        return new Promise(function (resolve, reject) {
          img.onload = resolve;
          img.onerror = reject;
        });
      }

      function readFileAsDataURL(file) {
        return new Promise(function (resolve, reject) {
          const reader = new FileReader();
          reader.onload = function () { resolve(reader.result); };
          reader.onerror = reject;
          reader.readAsDataURL(file);
        });
      }

      function drawCoverImage(ctx, img, targetWidth, targetHeight) {
        const sourceRatio = img.width / img.height;
        const targetRatio = targetWidth / targetHeight;
        let sx = 0;
        let sy = 0;
        let sw = img.width;
        let sh = img.height;

        if (sourceRatio > targetRatio) {
          sw = img.height * targetRatio;
          sx = (img.width - sw) / 2;
        } else {
          sh = img.width / targetRatio;
          sy = (img.height - sh) / 2;
        }

        ctx.drawImage(img, sx, sy, sw, sh, 0, 0, targetWidth, targetHeight);
      }

      async function compressImageToPassport(file) {
        const dataUrl = await readFileAsDataURL(file);
        const img = new Image();
        img.src = dataUrl;
        await waitForImageLoad(img);

        const canvas = document.createElement('canvas');
        canvas.width = 487;
        canvas.height = 600;

        const ctx = canvas.getContext('2d');
        ctx.fillStyle = '#ffffff';
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        drawCoverImage(ctx, img, canvas.width, canvas.height);
        return canvas.toDataURL('image/jpeg', 0.82);
      }

      async function onPhotoChange() {
        const file = photoInput.files && photoInput.files[0];
        if (!file) {
          photoDataUrl = '';
          refreshPhotoUploadState();
          scheduleRender();
          return;
        }

        photoPromise = compressImageToPassport(file)
          .then(function (dataUrl) {
            photoDataUrl = dataUrl;
            refreshPhotoUploadState();
            scheduleRender();
          })
          .catch(function () {
            photoDataUrl = '';
            refreshPhotoUploadState();
            photoInput.classList.add('is-error');
            setStatus('Photo upload failed. Please try a different image file.', 'is-error');
          })
          .finally(function () {
            photoPromise = null;
          });

        await photoPromise;
      }

      function buildPdfFileName() {
        return 'Biodata-MeghalayaJobs.in.pdf';
      }

      async function generatePDF() {
        if (photoPromise) await photoPromise;
        if (!validateForm()) return;

        if (!window.jspdf || !window.jspdf.jsPDF) {
          setStatus('The PDF library did not load correctly. Please refresh the page and try again.', 'is-error');
          return;
        }

        setLoadingState(true, 'Generating Your Bio-Data', 'Your PDF will download automatically when it is ready.');
        setStatus('Generating PDF...', 'is-success');

        try {
          const loadingStartedAt = Date.now();
          await waitForNextPaint();
          const result = renderPdfDocument(getFormData());
          const remainingDelay = Math.max(0, 3000 - (Date.now() - loadingStartedAt));
          if (remainingDelay) {
            await waitForMilliseconds(remainingDelay);
          }
          invalidateGeneratedPdf();
          generatedPdfUrl = URL.createObjectURL(result.blob);
          generatedPdfName = buildPdfFileName();
          triggerPdfDownload();
          setPrimaryButtonMode('download');
          setStatus(
            result.notes.length
              ? 'PDF generated successfully. Download started automatically. ' + result.notes.join(' ')
              : 'PDF generated successfully. Download started automatically.',
            'is-success'
          );
          downloadRow.classList.remove('is-visible');
        } finally {
          setLoadingState(false);
        }
      }

      function downloadGeneratedPdf() {
        if (!generatedPdfUrl) {
          setStatus('Please generate the PDF first.', 'is-error');
          return;
        }
        triggerPdfDownload();
      }

      function resetForm() {
        if (!window.confirm('Clear all fields and start again?')) return;

        $$('input, select, textarea').forEach(function (field) {
          field.value = '';
        });

        photoDataUrl = '';
        photoPromise = null;
        eduBox.innerHTML = '';
        otherBox.innerHTML = '';
        clearErrors();
        clearStatus();
        invalidateGeneratedPdf();
        refreshPhotoUploadState();
        closeCalendar();

        document.getElementById('jmrm-date').value = getTodayInputValue();
        declarationCheckbox.checked = false;

        addEducationRow();
      }

      root.addEventListener('click', function (event) {
        const calendarInput = event.target.closest('[data-calendar-input]');
        if (calendarInput instanceof HTMLInputElement) {
          openCalendar(calendarInput);
          return;
        }

        if (event.target === calendarOverlay) {
          closeCalendar();
          return;
        }

        const actionTarget = event.target.closest('[data-action]');
        if (!actionTarget) return;

        const action = actionTarget.getAttribute('data-action');

        if (action === 'calendar-prev') {
          calendarState.visibleMonth = new Date(
            calendarState.visibleMonth.getFullYear(),
            calendarState.visibleMonth.getMonth() - 1,
            1
          );
          renderCalendar();
          return;
        }

        if (action === 'calendar-next') {
          calendarState.visibleMonth = new Date(
            calendarState.visibleMonth.getFullYear(),
            calendarState.visibleMonth.getMonth() + 1,
            1
          );
          renderCalendar();
          return;
        }

        if (action === 'calendar-day') {
          const selectedDate = parseInputDate(actionTarget.getAttribute('data-date'));
          if (selectedDate) {
            setCalendarDate(selectedDate);
          }
          return;
        }

        if (action === 'calendar-clear') {
          clearCalendarDate();
          return;
        }

        if (action === 'calendar-today') {
          setCalendarDate(new Date());
          return;
        }

        if (action === 'calendar-close') {
          closeCalendar();
          return;
        }

        if (action === 'add-edu') {
          addEducationRow();
          scheduleRender();
          return;
        }

        if (action === 'add-other') {
          addOtherRow();
          scheduleRender();
          return;
        }

        if (action === 'remove-row') {
          const row = actionTarget.closest('.jmrm-row');
          if (row) row.remove();
          renumberRows(eduBox, 'Qualification');
          renumberRows(otherBox, 'Qualification');
          scheduleRender();
          return;
        }

        if (action === 'generate') {
          generatePDF().catch(function () {
            setStatus('Something went wrong while generating the PDF. Please try again.', 'is-error');
          });
          return;
        }

        if (action === 'download-ready') {
          downloadGeneratedPdf();
          return;
        }

        if (action === 'reset') {
          resetForm();
          return;
        }
      });
